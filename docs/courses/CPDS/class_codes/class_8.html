def sorted_seq(seq_of_digits):
  ans_list = []
  ans_list.append(seq_of_digits[0])
  best_list = []
  for ele in seq_of_digits[1:]:
    if ele>ans_list[-1]:
      ans_list.append(ele)
    else:
      best_list = ans_list
      ans_list = [ele]
    if len(ans_list)>len(best_list):
      best_list = ans_list
  return best_list
